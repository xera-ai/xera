import { select } from '@clack/prompts';

export type ProjectShape = 'web' | 'api' | 'mixed';

export async function askShape(defaultShape: ProjectShape = 'web'): Promise<ProjectShape> {
  const result = await select({
    message: 'What kind of testing does this project do?',
    initialValue: defaultShape,
    options: [
      { value: 'web', label: 'Web UI only (Playwright browser tests)' },
      { value: 'api', label: 'HTTP API only (REST/GraphQL endpoint tests, no browser)' },
      { value: 'mixed', label: 'Both (some UI tickets, some API tickets, in one repo)' },
    ],
  });
  if (typeof result === 'symbol') throw new Error('Cancelled');
  return result as ProjectShape;
}
```

// packages/cli/src/checks.ts (add)
import { join } from 'node:path';
import { existsSync } from 'node:fs';
import { readAuthState } from '@xera-ai/core';

export interface AuthFileCheckResult {
  role: string;
  ok: boolean;
  severity: 'info' | 'warn' | 'error';
  message: string;
  expiresInMs?: number;
}

export function checkHttpAuthFiles(input: { authDir: string; roles: readonly string[] }): AuthFileCheckResult[] {
  const dir = join(input.authDir, 'http');
  return input.roles.map((role) => {
    const path = join(dir, `${role}.json`);
    if (!existsSync(path)) {
      return {
        role, ok: false, severity: 'error',
        message: `Auth file missing: ${path}. Run: bun run xera:auth-setup --role ${role}`,
      };
    }
    const entry = readAuthState(dir, role);
    if (!entry) {
      return {
        role, ok: false, severity: 'error',
        message: `Auth file unreadable: ${path}. Re-run xera:auth-setup --role ${role}`,
      };
    }
    const expiresInMs = new Date(entry.expires_at).getTime() - Date.now();
    if (expiresInMs <= 0) {
      return {
        role, ok: false, severity: 'warn',
        message: `Auth file expired: ${path}. Run: bun run xera:auth-setup --role ${role}`,
        expiresInMs,
      };
    }
    return {
      role, ok: true, severity: 'info',
      message: `Auth file present (expires in ${Math.round(expiresInMs / 60000)}m)`,
      expiresInMs,
    };
  });
}
```

import { existsSync } from 'node:fs';

export interface OpenApiCheckResult {
  ok: boolean;
  severity: 'info' | 'warn';
  message: string;
}

export async function checkOpenApi(spec: string | undefined): Promise<OpenApiCheckResult> {
  if (!spec) {
    return {
      ok: true,
      severity: 'warn',
      message: 'OpenAPI spec not configured (http.spec). CONTRACT_DRIFT detection and schema-derived edge cases are disabled.',
    };
  }
  if (spec.startsWith('http://') || spec.startsWith('https://')) {
    try {
      const res = await fetch(spec);
      if (!res.ok) return { ok: false, severity: 'warn', message: `OpenAPI URL returned ${res.status}` };
    } catch (e) {
      return { ok: false, severity: 'warn', message: `OpenAPI URL unreachable: ${(e as Error).message}` };
    }
    return { ok: true, severity: 'info', message: 'OpenAPI URL reachable' };
  }
  if (!existsSync(spec)) {
    return { ok: false, severity: 'warn', message: `OpenAPI file not found: ${spec}` };
  }
  return { ok: true, severity: 'info', message: 'OpenAPI file present' };
}
```

export function readAuthState(authDir: string, role: string): AuthStateEntry | null {
  const p = pathFor(authDir, role);
  if (existsSync(p)) {
    const txt = readFileSync(p, 'utf8');
    const plain = decrypt(txt, resolveAuthKey());
    return AuthStateEntrySchema.parse(JSON.parse(plain));

import { existsSync, mkdirSync, renameSync, readdirSync } from 'node:fs';
import { join } from 'node:path';

export function migrateLegacyAuthDir(authDir: string): { moved: string[] } {
  const webDir = join(authDir, 'web');
  if (!existsSync(authDir)) return { moved: [] };
  mkdirSync(webDir, { recursive: true });
  const moved: string[] = [];
  for (const f of readdirSync(authDir)) {
    if (f.endsWith('.json') && !['web', 'http'].includes(f)) {
      renameSync(join(authDir, f), join(webDir, f));
      moved.push(f);
    }
  }
  return { moved };
}
```
